\begin{longtable}{p{0.38\linewidth} p{0.55\linewidth}}
\toprule
\textbf{Variabile} & \textbf{Descrizione} \\
\midrule
\endhead
\multicolumn{2}{l}{\textit{PostgreSQL}} \\[2pt]
\texttt{POSTGRES\_USER}          & Username del database. \\
\texttt{POSTGRES\_PASSWORD}      & Password del database. \\
\texttt{POSTGRES\_DB}            & Nome del database. \\
\texttt{POSTGRES\_PORT}          & Porta esposta all'host (default: 5432). \\
\texttt{DATABASE\_URL}           & Connection string completa (usata dai servizi Python). \\
\midrule
\multicolumn{2}{l}{\textit{Object Storage (SeaweedFS / S3)}} \\[2pt]
\texttt{S3\_ENDPOINT}            & URL dell'S3 gateway di SeaweedFS (es.\ \texttt{http://seaweedfs:8333}). \\
\texttt{S3\_ACCESS\_KEY}         & Access key S3. \\
\texttt{S3\_SECRET\_KEY}         & Secret key S3. \\
\texttt{GCP\_BUCKET\_NAME}       & Nome del bucket (default: \texttt{dms-storage}). \\
\midrule
\multicolumn{2}{l}{\textit{Sicurezza}} \\[2pt]
\texttt{EMAIL\_ENCRYPTION\_KEY}  & Chiave Fernet (AES-128-CBC) per la cifratura delle credenziali IMAP. Generare con: \texttt{python -c "from cryptography.fernet import Fernet; print(Fernet.generate\_key().decode())"}. \\
\texttt{SECRET\_KEY}             & Chiave HS256 per la firma dei JWT. \\
\midrule
\multicolumn{2}{l}{\textit{LLM e Agent Worker}} \\[2pt]
\texttt{MODEL\_NAME}             & Modello LLM target (es.\ \texttt{claude-sonnet-4-6}). \\
\texttt{BASE\_URL}               & Endpoint OpenAI-compatible (es.\ \texttt{https://api.anthropic.com}). \\
\texttt{CUSTOM\_API\_KEY}        & API key del provider LLM. \\
\texttt{AUTO\_FILE\_THRESHOLD}   & Soglia di confidenza per auto-filing (default: \texttt{0.85}). \\
\texttt{POLL\_INTERVAL\_SECONDS} & Intervallo di polling del worker in secondi (default: \texttt{30}). \\
\texttt{TEXT\_PREVIEW\_CHARS}    & Lunghezza massima dell'anteprima testuale inviata all'LLM (default: \texttt{2000}). \\
\midrule
\multicolumn{2}{l}{\textit{MCP Server}} \\[2pt]
\texttt{MCP\_SERVER\_URL}        & URL SSE del MCP server (es.\ \texttt{http://mcp\_server:8001/sse}). \\
\midrule
\multicolumn{2}{l}{\textit{OpenSearch}} \\[2pt]
\texttt{OPENSEARCH\_URL}         & URL del cluster OpenSearch (es.\ \texttt{http://opensearch:9200}). \\
\texttt{EMBEDDING\_MODEL\_NAME}  & Modello sentence-transformers per embedding (default: \texttt{sentence-transformers/paraphrase-multilingual-MiniLM-L12-v2}). \\
\midrule
\multicolumn{2}{l}{\textit{Frontend}} \\[2pt]
\texttt{VITE\_API\_URL}          & URL del backend esposta al browser (es.\ \texttt{http://localhost:8000}). Iniettata come build arg Docker. \\
\bottomrule
